## docker images build command
for svc in auth eta gateway ingester fleet attendance asynctask notifier report; do
  docker buildx build \
    --platform linux/amd64 \
    --build-arg BINARY_PATH=bin/$svc \
    -t ${svc}-stage:v1 \
    -f scripts/Dockerfile \
    --load \
    .
done
